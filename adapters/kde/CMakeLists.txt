cmake_minimum_required(VERSION 3.25)
project(appfacesthumbnail VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui)
find_package(KF6 6.0 REQUIRED COMPONENTS KIO CoreAddons)
set(APP_FACES_HELPER "${CMAKE_CURRENT_SOURCE_DIR}/thumbnail-helper" CACHE FILEPATH "App Faces thumbnail helper")
configure_file(config.h.in config.h @ONLY)
kcoreaddons_add_plugin(appfacesthumbnail SOURCES appfacesthumbnail.cpp INSTALL_NAMESPACE "kf6/thumbcreator")
target_include_directories(appfacesthumbnail PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(appfacesthumbnail PRIVATE Qt6::Core Qt6::Gui KF6::KIOGui KF6::CoreAddons)
include(CTest)
if(BUILD_TESTING)
  add_executable(appfaces_kde_smoke tests/smoke.cpp)
  target_link_libraries(appfaces_kde_smoke PRIVATE Qt6::Core Qt6::Gui KF6::KIOGui KF6::CoreAddons)
  add_test(NAME kde_plugin_roundtrip COMMAND appfaces_kde_smoke $<TARGET_FILE:appfacesthumbnail>)
endif()
